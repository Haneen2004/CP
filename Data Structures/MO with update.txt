int S=2714; // = (2*n*n)^(1/3) 
// time complexty q*S
struct query{
    int l,r,idx,t;
    bool operator<(query&o){
        if(l/S!=o.l/S){
            return l/S<o.l/S;
        }
        if(r/S!=o.r/S){
            return r/S<o.r/S;
        }
        return t<o.t;
    }
};
struct upds{
    int idx,oldV,newV;
};
int main() {
    vector<query>qs;
    vector<upds>up;
    sort(all(qs));
    auto delElem=[&](int i){};
    auto addElem=[&](int i){};
    int l=0,r=-1,t=-1;
    auto addUpd=[&](int i){};
    auto delUpd=[&](int i){};
    auto getAns=[&](){ };
    for(auto[lq,rq,i,tq]:qs){
        while (t+1<tq) addUpd(++t);
        while (t>=tq) delUpd(t--);
        while (r < rq) addElem(++r);
        while (l > lq) addElem(--l);
        while (r > rq) delElem(r--);
        while (l < lq) delElem(l++);
        ans[i]=getAns();
    }
}
