struct sparse{
    sparse(vector<ll>&v,ll n){
        sset(v,n);
    }
    vector<vector<ll>>T;
    void sset(vector<ll>&v,ll n){
        T=vector<vector<ll>>(n,vector<ll>(20));
        for(int i=0;i<n;++i){
            T[i][0]=v[i];
        }
        for(int pw=1;(1<<pw)<=n;++pw){
            for (int i = 0; i+(1<<pw) <=n ; ++i) {
                T[i][pw]=max(T[i][pw-1],T[i+(1<<(pw-1))][pw-1]);
            }
        }
    }
    ll query(ll l,ll r){
        if(r<l) swap(l,r);
        ll z=log2(r-l+1);
        return max(T[l][z],T[r-(1<<z)+1][z]);
    }
};

//  q o(1)
struct Sparse {
    int n;
    vector<int> lg;
    vector<vector<int>> st;

    Sparse() {}

    Sparse(vector<int> &v) {
        build(v);
    }

    void build(vector<int> &v) {
        n = v.size();

        lg.assign(n + 1, 0);
        for (int i = 2; i <= n; i++)
            lg[i] = lg[i / 2] + 1;

        int K = lg[n] + 1;
        st.assign(K, vector<int>(n));

        for (int i = 0; i < n; i++)
            st[0][i] = v[i];

        for (int j = 1; j < K; j++) {
            for (int i = 0; i + (1 << j) <= n; i++) {
                st[j][i] = min(st[j - 1][i],
                               st[j - 1][i + (1 << (j - 1))]);
            }
        }
    }

    int query(int l, int r) {
        if (l > r) return LLONG_MAX;
        int j = lg[r - l + 1];
        return min(st[j][l], st[j][r - (1 << j) + 1]);
    }
};
