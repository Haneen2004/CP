string addStrings(string a, string b) {
    string res;
    int carry = 0, i = a.size() - 1, j = b.size() - 1;
    while(i >= 0 || j >= 0 || carry) {
        int x = i >= 0 ? a[i--] - '0' : 0;
        int y = j >= 0 ? b[j--] - '0' : 0;
        int sum = x + y + carry;
        res += (sum % 10) + '0';
        carry = sum / 10;
    }
    reverse(res.begin(), res.end());
    return res;
}

string subtractStrings(string a, string b) {
    string res;
    int borrow = 0, i = a.size() - 1, j = b.size() - 1;
    while(i >= 0) {
        int x = a[i] - '0' - borrow;
        int y = j >= 0 ? b[j] - '0' : 0;

        if(x < y) {
            x += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        res += (x - y) + '0';
        i--; j--;
    }
    while(res.size() > 1 && res.back() == '0') res.pop_back();
    reverse(res.begin(), res.end());
    return res;
}


string multiplyStrings(string a, string b) {
    int n = a.size(), m = b.size();
    vector<int> res(n + m, 0);
    for(int i = n - 1; i >= 0; i--) {
        for(int j = m - 1; j >= 0; j--) {
            int mul = (a[i]-'0') * (b[j]-'0');
            int sum = mul + res[i+j+1];

            res[i+j+1] = sum % 10;
            res[i+j] += sum / 10;
        }
    }
    string result;
    for(int digit : res) {
        if(!(result.empty() && digit == 0)) result += digit + '0';
    }
    return result.empty() ? "0" : result;
}
string divideStrings(string a, string b) {
    if (b == "0") throw invalid_argument("Division by zero");
    string result, curr;
    for (char c : a) {
        curr += c;
        int x = 0;
        while (curr.size() > 1 && curr[0] == '0') curr.erase(0, 1);
        while (curr.size() > b.size() || (curr.size() == b.size() && curr >= b)) {
            curr = subtractStrings(curr, b);
            x++;
        }
        result += x + '0';
    }
    // Remove leading zeros
    int i = 0;
    while (i < result.size() - 1 && result[i] == '0') i++;
    return result.substr(i);
}
