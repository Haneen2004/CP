struct LCA{
    int n;
    vector<int>d;
    vector<vector<int>>up;
    LCA(const vector<vector<int>>&adj,int rt=1){
        n=sz(adj);
        d=vector<int>(n);
        up=vector<vector<int>>(n,vector<int>(30));
        dfs(rt,0,adj);
    }
    void dfs(ll u,ll p,const vector<vector<int>>&adj){
        d[u]=d[p]+1;
        for(auto&i:adj[u]){
            if(i==p) continue;
            up[i][0]=u;
            for (int j = 1; j <30 ; ++j) {
                up[i][j]=up[up[i][j-1]][j-1];
            }
            dfs(i,u,adj);
        }
    }
    int k_th(int u,int k){
        for (int i = 0; i <30 ; ++i) {
            if(k>>i&1)
                u=up[u][i];
        }
        return u;
    }
    int lca(int u,int v){
        int k;
        if(d[v]>d[u])
            swap(u,v);
        k=d[u]-d[v];
        u= k_th(u,k);
        if(u==v) return u;
        for(int i=29;i>=0;--i){
            if(up[u][i]!=up[v][i])
                u=up[u][i],v=up[v][i];
        }
        return up[u][0];
    };
};
