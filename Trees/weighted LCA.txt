struct W_LCA{
    int n,natural;
    vector<int>d;
    vector<vector<int>>up,mx;
    int merge(int x,int y ){
        return max(x,y);
    }
    W_LCA(const vector<vector<pair<int,int>>>&adj,int rt=1){
        natural=0;
        n=sz(adj);
        d=vector<int>(n);
        up=vector<vector<int>>(n,vector<int>(30));   mx=vector<vector<int>>(n,vector<int>(30,natural));
        dfs(rt,0,adj);
    }
    void dfs(ll u,ll p,const vector<vector<pair<int,int>>>&adj){
        d[u]=d[p]+1;
        for(auto&[i,w]:adj[u]){
            if(i==p) continue;
            up[i][0]=u;
            mx[i][0]=w;
            for (int j = 1; j <30 ; ++j) {
                up[i][j]=up[up[i][j-1]][j-1];
                mx[i][j]= merge(mx[i][j-1],mx[up[i][j-1]][j-1]);
            }
            dfs(i,u,adj);
        }
    }
    int kth_node(int u,int k){
        for (int i = 0; i <30 ; ++i) {
            if(k>>i&1)
                u=up[u][i];
        }
        return u;
    }
    int kth_val(int u,int k){
        ll ret=natural;
        for (int i = 0; i <30 ; ++i) {
            if(k>>i&1) {
                ret = merge(ret, mx[u][i]);
                u = up[u][i];
            }
        }
        return ret;
    }
    int lca(int u,int v){
        int k;
        if(d[v]>d[u])
            swap(u,v);
        k=d[u]-d[v];
        u= kth_node(u,k);
        if(u==v) return u;
        for(int i=29;i>=0;--i){
            if(up[u][i]!=up[v][i])
                u=up[u][i],v=up[v][i];
        }
        return up[u][0];
    };
};
