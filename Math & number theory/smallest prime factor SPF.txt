vector<int> spf(N);
void SPF() {
    for (int i = 0; i < N; ++i) {
        spf[i] = i;
    }
    for (int i = 2; i * i < N; ++i) {
        if (spf[i] == i) {
            for (int j = i * i; j < N; j += i) {
                if (spf[j] == j)
                    spf[j] = i;
            }
        }
    }
}
