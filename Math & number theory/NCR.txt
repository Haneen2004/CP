// nCr    loop over r
ll ncr(ll n, ll r) {
    if (r > n) return 0;
    if (r == 0 || r == n) return 1;
    r = min(r, n - r);
    ll ret = 1;
    for (ll i = 1; i <= r; i++) {
        ret = ret * (n - r + i) % mod;
        ret = ret * pow_mod(i, mod - 2, mod) % mod;
    }
    return ret;
}
