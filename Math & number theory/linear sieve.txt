vector<bool> isprime(N, 1);
vector<int> prime;
void l_sieve() {
    isprime[0] = isprime[1] = 0;
    for (int i = 2; i < N; ++i) {
        if (isprime[i]) prime.push_back(i);
        for (auto it: prime) {
            if (i * it >= N) break;
            isprime[i * it] = 0;
            if (i % it == 0) break;
        }
    }
}
