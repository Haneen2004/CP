string bigIntMul(string&a,string&b){
    int n=sz(a),m=sz(b);
    vi p1(n),p2(m);
    for (int i = 0; i < n ; ++i) {
        p1[i]=a[n-i-1]-'0';
    }
    for (int i = 0; i < m ; ++i) {
        p2[i]=b[m-i-1]-'0';
    }

    vi mul= multiply(p1,p2);
    string ret;
    for (int i = 0; i+1 < sz(mul) ; ++i) {
        mul[i+1]+=mul[i]/10;
        mul[i]%=10;
        ret+=char('0'+mul[i]);
    }
    while (sz(ret)>1&&ret.back()=='0') ret.pop_back();
    reverse(all(ret));
    return ret;
}
