// 2 tangents, out -> two pairs of points for each of the tangents
// 1 tangent,  out -> P 4 times (tangent line can be found as line(o1,p).perpThrough(p) )
// 0 tangents, it does nothing;
// circles are identical, it aborts.
//same code can find tangent to a circle passing through a point (by setting r2 to 0 inner doesnt matter).
int tangents(pt o1, double r1, pt o2, double r2, bool inner, vector<pair<pt,pt>> &out) {
    if (inner) r2 =-r2;
    pt d = o2-o1;
    double dr = r1-r2, d2 = sq(d), h2 = d2-dr*dr;
    if (d2 == 0 || h2 < 0) {assert(h2 != 0); return 0;}
    for (double sign : {-1,1}) {
        pt v = (d*dr + perp(d)*sqrt(h2)*sign)/d2;
        out.push_back({o1 + v*r1, o2 + v*r2});
    }
    return 1 + (h2 > 0);
}
