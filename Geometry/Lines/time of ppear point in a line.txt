// get t from point
 auto getT=[&](pt p){
     pt AB = B - A,AP = p - A;
     return (dot(AP, AB) / dot(AB, AB));
};
// get point from t
 auto getP=[&](T t){
    pt p=A+t*(B-A);
    return p;
 };
