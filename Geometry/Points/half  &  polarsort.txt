//Polar sort (sort point by angle they make around point)
bool half(pt p) { // true if in blue half
    assert(p.x != 0 || p.y != 0); // the argument of (0,0) is undefined
    return p.y > 0 || (p.y == 0 && p.x < 0);
}
//the starting angle of the ordering is v
pt v = {/* whatever you want except 0,0 */};
bool half(pt p) {
    return cross(v,p) < 0 || (cross(v,p) == 0 && dot(v,p) < 0);
}
void polarSort(vector<pt> &v) {
    sort(v.begin(), v.end(), [](pt v, pt w) {
        return make_tuple(half(v), 0) <
               make_tuple(half(w), cross(v,w));
    });
}
//if 2 points in the same direction,the point is further from the origin will appear later
void polarSort(vector<pt> &v) {
    sort(v.begin(), v.end(), [](pt v, pt w) {
        return make_tuple(half(v), 0, sq(v)) <
               make_tuple(half(w), cross(v,w), sq(w));
    });
}
//polar sort around some point O other than the origin
void polarSortAround(pt o, vector<pt> &v) {
    sort(v.begin(), v.end(), [&o](pt v, pt w) {
        return make_tuple(half(v - o), 0) < make_tuple(half(w - o), cross(v - o, w - o));
    });
}
