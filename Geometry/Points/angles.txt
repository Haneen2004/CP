// check if two vectors are perpendicular
bool isPerp(pt v, pt w) {return dot(v,w) == 0;}


// angle between 2 vectors
long double angle(pt v, pt w) {
    double cosTheta = dot(v,w) / abs(v) / abs(w);
    return acos(max(-1.0, min(1.0, cosTheta)));
    // or form c++17
    return acos(clamp(dot(v,w) / abs(v) / abs(w),-1.0l, 1.0l));
}

//  check if point P lies in the angle formed by lines AB and AC
// orient(a,b,c) can't be zero
bool inAngle(pt a, pt b, pt c, pt p) {
    assert(orient(a,b,c) != 0);
    if (orient(a,b,c) < 0) swap(b,c);
    return orient(a,b,p) >= 0 && orient(a,c,p) <= 0;
}

// angle BAC
long double orientedAngle(pt a, pt b, pt c) {
    if (orient(a,b,c) >= 0)
        return angle(b-a, c-a);
    else
        return 2*M_PI- angle(b-a, c-a);
}

// amplitude travelled around point A, from P to Q
long double angleTravelled(pt a, pt p, pt q) {
    long double ampli = angle(p-a, q-a);
    if (orient(a,p,q) > 0) return ampli;
    else return -ampli;
}
